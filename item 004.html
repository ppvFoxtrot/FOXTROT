VECTOR - Personal Performance & Progress Telemetry System

VECTOR — Personal Performance & Progress Telemetry System

System overview:

VECTOR is a Python-based telemetry and performance-tracking engine built to quantify daily output, study consistency, and health habits into actionable data. Operating over a local SQLite relational backend vector_data.db, the system computes a dynamic daily Master Score out of 100 by aggregating configurable core categories—such as Academics, FOXTROT/Career Projects, and Sleep—while factoring in weighted negative debuffs.

Technical Architecture & Capabilities:

The software architecture decouples configuration, evaluation, and visualization across modular components:

  • Analytical Engine: Computes historical moving averages over 7-day, 30-day, and 365-day periods, tracks consecutive logging streaks, and calculates comparative weekly performance deltas.
  • Dynamic Scoring Pipeline: Normalizes inputs across 0-10 standard and 0-5 multiplier scales, applying custom weight coefficients and mapping total scores to qualitative performance thresholds ranging from CRITICAL to EXCELLENT
  • Desktop Presentation: Features a modular Tkinter GUI complete with integrated Matplotlib trend visualizations, an interactive monthly calendar log, custom input forms, and dynamic settings management.

    from datetime import datetime, timedelta


class AnalyticsEngine:

    def __init__(self, db):
        self.db = db

    def get_all_records_chronological(self):
        records = self.db.get_all_entries()
        # Ensure chronological ordering
        records.sort(
            key=lambda x: datetime.strptime(x[0], "%Y-%m-%d").date()
            if isinstance(x[0], str)
            else x[0]
        )
        return records

    def compute_dashboard_stats(self):
        records = self.get_all_records_chronological()

        if not records:
            return {
                "total_days": 0,
                "current_streak": 0,
                "weekly_avg": 0.0,
                "monthly_avg": 0.0,
                "yearly_avg": 0.0,
                "best_day": None,
                "worst_day": None,
                "curr_7d_avg": 0.0,
                "prev_7d_avg": 0.0,
                "delta_7d": 0.0,
            }

        scores_by_date = {r[0]: r[1] for r in records}
        today_str = datetime.now().strftime("%Y-%m-%d")

        # Basic Stats
        total_days = len(records)
        scores = [r[1] for r in records]

        best_score = max(scores)
        worst_score = min(scores)

        best_day = next(r for r in records if r[1] == best_score)
        worst_day = next(r for r in records if r[1] == worst_score)

        # Streak calculation
        curr_streak = 0
        check_date = datetime.now().date()

        # Allow today to be unlogged without breaking current streak from yesterday
        if check_date.strftime("%Y-%m-%d") not in scores_by_date:
            check_date -= timedelta(days=1)

        while check_date.strftime("%Y-%m-%d") in scores_by_date:
            curr_streak += 1
            check_date -= timedelta(days=1)

        # Time Window Averages
        now = datetime.now().date()
        d_7 = (now - timedelta(days=7)).strftime("%Y-%m-%d")
        d_30 = (now - timedelta(days=30)).strftime("%Y-%m-%d")
        d_365 = (now - timedelta(days=365)).strftime("%Y-%m-%d")

        w_scores = [r[1] for r in records if r[0] >= d_7]
        m_scores = [r[1] for r in records if r[0] >= d_30]
        y_scores = [r[1] for r in records if r[0] >= d_365]

        weekly_avg = sum(w_scores) / len(w_scores) if w_scores else 0.0
        monthly_avg = sum(m_scores) / len(m_scores) if m_scores else 0.0
        yearly_avg = sum(y_scores) / len(y_scores) if y_scores else 0.0

        # Trend Analysis: Last 7 days vs Prior 7 days
        curr_7d = [
            r[1]
            for r in records
            if (now - timedelta(days=7))
            <= datetime.strptime(r[0], "%Y-%m-%d").date()
            <= now
        ]
        prev_7d = [
            r[1]
            for r in records
            if (now - timedelta(days=14))
            <= datetime.strptime(r[0], "%Y-%m-%d").date()
            < (now - timedelta(days=7))
        ]

        c_7_avg = sum(curr_7d) / len(curr_7d) if curr_7d else 0.0
        p_7_avg = sum(prev_7d) / len(prev_7d) if prev_7d else 0.0
        delta = c_7_avg - p_7_avg

        return {
            "total_days": total_days,
            "current_streak": curr_streak,
            "weekly_avg": round(weekly_avg, 1),
            "monthly_avg": round(monthly_avg, 1),
            "yearly_avg": round(yearly_avg, 1),
            "best_day": (best_day[0], round(best_day[1], 1)),
            "worst_day": (worst_day[0], round(worst_day[1], 1)),
            "curr_7d_avg": round(c_7_avg, 1),
            "prev_7d_avg": round(p_7_avg, 1),
            "delta_7d": round(delta, 1),
        }

class ConfigManager:

    def __init__(self, db):
        self.db = db
        self.categories = self.load_categories()
        self.thresholds = self.load_thresholds()

    def load_categories(self):
        raw = self.db.get_config("categories")
        if raw:
            try:
                loaded = json.loads(raw)
                # Ensure backward-compatibility with previously saved databases missing max_score
                for key, data in loaded.items():
                    if "max_score" not in data:
                        data["max_score"] = (
                            5 if key in ["other", "debuffs"] else 10
                        )
                return loaded
            except Exception:
                pass
        return DEFAULT_CATEGORIES.copy()

    def save_categories(self, categories):
        self.categories = categories
        self.db.save_config("categories", json.dumps(categories))

    def load_thresholds(self):
        raw = self.db.get_config("thresholds")
        if raw:
            try:
                return json.loads(raw)
            except Exception:
                pass
        return DEFAULT_THRESHOLDS.copy()

    def save_thresholds(self, thresholds):
        self.thresholds = thresholds
        self.db.save_config("thresholds", json.dumps(thresholds))
