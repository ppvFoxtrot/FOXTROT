class ConfigManager:

    def __init__(self, db):
        self.db = db
        self.categories = self.load_categories()
        self.thresholds = self.load_thresholds()

    def load_categories(self):
        raw = self.db.get_config("categories")
        if raw:
            try:
                loaded = json.loads(raw)
                # Ensure backward-compatibility with previously saved databases missing max_score
                for key, data in loaded.items():
                    if "max_score" not in data:
                        data["max_score"] = (
                            5 if key in ["other", "debuffs"] else 10
                        )
                return loaded
            except Exception:
                pass
        return DEFAULT_CATEGORIES.copy()

    def save_categories(self, categories):
        self.categories = categories
        self.db.save_config("categories", json.dumps(categories))

    def load_thresholds(self):
        raw = self.db.get_config("thresholds")
        if raw:
            try:
                return json.loads(raw)
            except Exception:
                pass
        return DEFAULT_THRESHOLDS.copy()

    def save_thresholds(self, thresholds):
        self.thresholds = thresholds
        self.db.save_config("thresholds", json.dumps(thresholds))

        entry = self.db.get_entry(today_str)

            score = entry["master_score"]
            cls = self.scoring.classify_score(score)
            self.lbl_dash_master.config(text=f"{score:.1f}")
            self.lbl_dash_class.config(text=cls)
        else:
            self.lbl_dash_master.config(text="--")

        self.lbl_dash_streak.config(
            text=f"Current Streak: {stats['current_streak']} Days"
        )
        self.lbl_dash_weekly.config(text=f"Weekly Avg: {stats['weekly_avg']}")

        delta = stats["delta_7d"]
        symbol = "▲" if delta > 0 else "▼" if delta < 0 else "▶"
        self.lbl_dash_trend.config(
            text=f"7d Trend vs Prior 7d: {symbol} {abs(delta):.1f} pts"
        )

        self.render_dashboard_graph()

    def render_dashboard_graph(self):
        for widget in self.chart_frame.winfo_children():
            widget.destroy()

        records = self.analytics.get_all_records_chronological()[-14:]

        fig, ax = plt.subplots(figsize=(7, 3), facecolor="#181825")
        ax.set_facecolor("#1e1e2e")

        if records:
            dates = [r[0][5:] for r in records]
            scores = [r[1] for r in records]

            ax.plot(
                dates,
                scores,
                marker="o",
                color="#89b4fa",
                linewidth=2,
                markersize=6,
            )
            ax.fill_between(dates, scores, color="#89b4fa", alpha=0.15)
            ax.set_ylim(0, 105)

        ax.tick_params(colors="#cdd6f4", labelsize=8)
        for spine in ax.spines.values():
            spine.set_color("#45475a")

        ax.set_title("14-Day Performance Trend", color="#cdd6f4", fontsize=10)

        canvas = FigureCanvasTkAgg(fig, master=self.chart_frame)
        canvas.draw()
        canvas.get_tk_widget().pack(fill="both", expand=True, padx=5, pady=5)

    def load_date_into_entry(self):
        d_str = self.ent_date.get().strip()
        data = self.db.get_entry(d_str)

        self.txt_notes.delete("1.0", tk.END)

        if not data:
            messagebox.showinfo(
                "Info", f"No record exists for {d_str}. Ready for new entry."
            )
            for w in self.cat_widgets.values():
                w["active"].set(True)
                w["entry"].delete(0, tk.END)
                w["entry"].insert(0, "0")
            return

        self.txt_notes.insert("1.0", data["notes"] or "")
        cats = data["categories"]

        for k, w in self.cat_widgets.items():
            if k in cats:
                w["active"].set(cats[k]["active"])
                w["entry"].delete(0, tk.END)
                w["entry"].insert(0, str(cats[k]["score"]))

    def save_entry(self):
        d_str = self.ent_date.get().strip()
        try:
            datetime.strptime(d_str, "%Y-%m-%d")
        except ValueError:
            messagebox.showerror(
                "Invalid Date", "Use YYYY-MM-DD date format."
            )
            return

        inputs = {}
        for k, w in self.cat_widgets.items():
            try:
                sc = float(w["entry"].get())
            except ValueError:
                messagebox.showerror(
                    "Input Error", f"Invalid numeric score in category key: {k}"
                )
                return

            inputs[k] = {"active": w["active"].get(), "score": sc}

        result = self.scoring.calculate_score(inputs)
        notes = self.txt_notes.get("1.0", tk.END).strip()

        self.db.save_entry(d_str, result["master_score"], notes, inputs)
        messagebox.showinfo(
            "Saved",
            f"Record saved successfully.\nMaster Score: {result['master_score']} ({result['classification']})",
        )

        self.refresh_dashboard()
        self.refresh_calendar()

    def delete_entry(self):
        d_str = self.ent_date.get().strip()
        if messagebox.askyesno(
            "Confirm Delete", f"Delete record for {d_str}?"
        ):
            self.db.delete_entry(d_str)
            self.refresh_dashboard()
            self.refresh_calendar()
            messagebox.showinfo("Deleted", "Record removed.")

    def prev_month(self):
        self.curr_month -= 1
        if self.curr_month < 1:
            self.curr_month = 12
            self.curr_year -= 1

    def next_month(self):
        self.curr_month += 1
        if self.curr_month > 12:
            self.curr_month = 1
            self.curr_year += 1
        self.refresh_calendar()

    def refresh_calendar(self):
        for w in self.cal_grid.winfo_children():
            w.destroy()

        self.lbl_cal_month.config(
            text=f"{calendar.month_name[self.curr_month]} {self.curr_year}"
        )

        days = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]
        for col, d in enumerate(days):
            lbl = ttk.Label(self.cal_grid, text=d, font=("Segoe UI", 9, "bold"))
            lbl.grid(row=0, column=col, padx=5, pady=5)

        month_matrix = calendar.monthcalendar(self.curr_year, self.curr_month)
        records = dict(self.db.get_all_entries())

        for row_idx, week in enumerate(month_matrix, start=1):
            for col_idx, day in enumerate(week):
                if day == 0:
                    continue

                d_str = f"{self.curr_year:04d}-{self.curr_month:02d}-{day:02d}"
                has_data = d_str in records

                btn_bg = "#313244"
                text = f"{day}"

                if has_data:
                    sc = records[d_str]
                    text += f"\n[{sc:.0f}]"
                    btn_bg = "#254a32" if sc >= 70 else "#5a2525"

                btn = tk.Button(
                    self.cal_grid,
                    text=text,
                    bg=btn_bg,
                    fg="#cdd6f4",
                    relief="flat",
                    command=lambda ds=d_str: self.select_cal_date(ds),
                )
                btn.grid(
                    row=row_idx, column=col_idx, sticky="nsew", padx=2, pady=2
                )

        for i in range(7):
            self.cal_grid.columnconfigure(i, weight=1)

class ScoringEngine:

    def __init__(self, config_manager):
        self.config_manager = config_manager

    def calculate_score(self, category_inputs):
        """Calculates performance breakdown based on 0-10 and 0-5 scale inputs.

        category_inputs format: {
            'cat_key': {'active': True/False, 'score': float}
        }
        """
        categories = self.config_manager.categories

        pos_earned = 0.0
        neg_penalty = 0.0
        breakdown = {}

        core_pos_categories = [
            k
            for k, v in categories.items()
            if float(v.get("max_score", 10)) == 10 and v["type"] == "positive"
        ]
        total_core_weight = sum(
            categories[k]["weight"] for k in core_pos_categories
        )

        for cat_key, cat_info in categories.items():
            input_data = category_inputs.get(
                cat_key, {"active": False, "score": 0.0}
            )
            is_active = input_data["active"]
            max_sc = float(cat_info.get("max_score", 10))
            raw_score = (
                max(0.0, min(max_sc, float(input_data["score"])))
                if is_active
                else 0.0
            )

            weight = float(cat_info["weight"])
            cat_type = cat_info["type"]

            if max_sc == 5:
                # Per-point weight multiplier scale (0 to 5)
                contribution = raw_score * weight if is_active else 0.0
                if cat_type == "positive":
                    pos_earned += contribution
                else:
                    contribution = -contribution
                    neg_penalty += abs(contribution)
            else:
                # Scaled weight calculation for standard 0-10 scale
                effective_weight = (
                    (weight / total_core_weight * 100)
                    if total_core_weight > 0
                    else 0
                )
                contribution = (
                    (raw_score / 10.0) * effective_weight if is_active else 0.0
                )
                pos_earned += contribution

            breakdown[cat_key] = {
                "name": cat_info["name"],
                "active": is_active,
                "raw_score": raw_score,
                "max_score": max_sc,
                "contribution": round(contribution, 2),
                "type": cat_type,
            }

        final_master = max(0.0, min(100.0, pos_earned - neg_penalty))
        classification = self.classify_score(final_master)

        return {
            "master_score": round(final_master, 2),
            "positive_score": round(pos_earned, 2),
            "penalty_score": round(neg_penalty, 2),
            "classification": classification,
            "breakdown": breakdown,
        }

    def classify_score(self, score):
        thresholds = self.config_manager.thresholds
        for label, bounds in thresholds.items():
            low, high = bounds
            if low <= score <= high:
                return label
        return "UNCLASSIFIED"
